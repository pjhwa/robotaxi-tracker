# ============================================================
# API BASE URL
# ============================================================
API_BASE = "https://txmccs.txdmv.gov/api/TruckStop"
#
# BREAKING CHANGE (2026-07-30):
# Old endpoints /operators/{auth} and /operators/{auth}/vehicles now return SPA HTML.
# Public read path is company-centric (businessEntityId).

# ============================================================
# SEARCH / LIST OPERATORS
# ============================================================
# Endpoint: GET /api/TruckStop/companies
# Query params:
#   searchType = "company_name" | "autonomous_vehicle_authorization_number"
#                | "address" | "usdot"
#   searchValue = search string
#
# Example:
#   GET .../companies?searchType=company_name&searchValue=Tesla
#   GET .../companies?searchType=autonomous_vehicle_authorization_number&searchValue=AV8313426653583
#
# Response shape (2026-07-30+):
# {
#   "results": [
#     {
#       "businessEntityId": "81edcff1-...",
#       "companyName": "Tesla Robotaxi, LLC, DBA: Tesla Robotaxi, LLC",
#       "autonomousVehicleAuthorizationNumber": "AV8313426653583",  // only if AV
#       "autonomousVehicleStatus": "authorized",
#       "physicalAddress": { ... },
#       "mailingAddress": { ... }
#     }
#   ],
#   "total": 3
# }
#
# NOTE: Many non-AV motor carriers appear in company_name search.
# Filter to rows that have autonomousVehicleAuthorizationNumber.
# Seed known auth numbers via autonomous_vehicle_authorization_number search —
# name search alone can miss some operators (e.g. Aurora rebrand).

SEARCH_OPERATORS_URL = "https://txmccs.txdmv.gov/api/TruckStop/companies?searchType=company_name&searchValue={query}"

COMPANY_DETAIL_URL = "https://txmccs.txdmv.gov/api/TruckStop/companies/{businessEntityId}"

OPERATOR_VEHICLES_URL = "https://txmccs.txdmv.gov/api/TruckStop/companies/{businessEntityId}/automated-motor-vehicles"

# ============================================================
# DATA FIELD MAPPINGS (from API JSON, not DOM)
# ============================================================
# Operator name:      company["companyName"]  (strip ", DBA: ...")
# Authorization #:    company["autonomousVehicleAuthorizationNumber"]
# Status:             company["autonomousVehicleStatus"]  ("authorized" | "not_authorized")
# Business entity:    company["businessEntityId"]
# Vehicle count:      len(vehicles_response["vehicles"])
# Vehicle make:       vehicles_response["vehicles"][n]["make"]
# Vehicle model:      vehicles_response["vehicles"][n]["model"]
# Vehicle model year: vehicles_response["vehicles"][n]["modelYear"]
# Vehicle VIN:        vehicles_response["vehicles"][n]["vin"]
